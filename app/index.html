if (riskPosture <= 2) weighted.compliance += 2; if (riskPosture >= 4 && operations >= 3) weighted.auditing += 2; if (automation <= 2 && operations >= 3) weighted.automation += 2; if (clarity >= 4 && operations >= 3) weighted.scaling += 2;
